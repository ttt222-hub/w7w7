local function buildFontSpec(entry)
    if entry.Family then
        local ok, font = pcall(function()
            return Font.new("rbxasset://fonts/families/" .. entry.Family .. ".json")
        end)
        if ok and font then return { Face = font } end
    end
    return { Enum = entry.Fallback }
end

local trackedTextObjects = {}
local function registerText(obj)
    if obj and (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) then
        table.insert(trackedTextObjects, obj)
    end
    return obj
end

local function applyFont(obj, fontSpec)
    if not obj or not fontSpec then return end
    if fontSpec.Face then
        pcall(function() obj.FontFace = fontSpec.Face end)
    elseif fontSpec.Enum then
        pcall(function() obj.Font = fontSpec.Enum end)
    end
end

local function applyFontToAll()
    for _, obj in ipairs(trackedTextObjects) do
        if obj and obj.Parent then
            applyFont(obj, currentFontSpec)
        end
    end
end

local function new(class, parent, props)
    local inst = Instance.new(class)
    if props then
        for k, v in pairs(props) do
            pcall(function() inst[k] = v end)
        end
    end
    if parent then inst.Parent = parent end
    registerText(inst)
    return inst
end

local function makeDraggable(frame, dragArea)
    dragArea = dragArea or frame
    local dragging, dragStart, startPos
    dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            local conn
            conn = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if conn then conn:Disconnect() end
                end
            end)
        end
    end)
    dragArea.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                local delta = input.Position - dragStart
                frame.Position = UDim2.new(
                    startPos.X.Scale, startPos.X.Offset + delta.X,
                    startPos.Y.Scale, startPos.Y.Offset + delta.Y
                )
            end
        end
    end)
end

local function getGameFontSpec()
    for _, entry in ipairs(FONT_LIST) do
        if entry.Name == (env.w7w7_GameFont or "Michroma") then
            return buildFontSpec(entry)
        end
    end
    return buildFontSpec(FONT_LIST[1])
end

local ICON_FONT_STR = tostring(Font.new("rbxasset://LuaPackages/Packages/_Index/BuilderIcons/BuilderIcons/BuilderIcons.json"))

local function isIconText(obj)
    local ok, str = pcall(function() return tostring(obj.FontFace) end)
    if ok and str then
        if str == ICON_FONT_STR then return true end
        if string.find(str, "BuilderIcons") then return true end
    end
    return false
end

local function isOurPanel(obj)
    return sg and obj:IsDescendantOf(sg)
end

local function trySetFont(obj)
    if not (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) then
        return false
    end
    if isIconText(obj) then return false end
    if isOurPanel(obj) then return false end
    local saved = gameFontOriginals[obj]
    if not saved then
        saved = { FontFace = obj.FontFace, Font = obj.Font }
        gameFontOriginals[obj] = saved
    end
    local before = tostring(obj.FontFace)
    if gameFontSpec.Face then
        pcall(function() obj.FontFace = gameFontSpec.Face end)
    elseif gameFontSpec.Enum then
        pcall(function() obj.Font = gameFontSpec.Enum end)
    end
    return before ~= tostring(obj.FontFace)
end

local function restoreGameFont(obj)
    local saved = gameFontOriginals[obj]
    if not saved then return end
    pcall(function() obj.FontFace = saved.FontFace end)
    pcall(function() obj.Font = saved.Font end)
    gameFontOriginals[obj] = nil
end

local function getAllUIRoots()
    local roots = { game }
    pcall(function()
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        if pg then table.insert(roots, pg) end
    end)
    pcall(function()
        local cg = game:GetService("CoreGui")
        if cg then table.insert(roots, cg) end
    end)
    pcall(function()
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        local mainGui = pg and pg:FindFirstChild("MainGui")
        if mainGui then table.insert(roots, mainGui) end
    end)
    return roots
end

local function scanGameFonts()
    task.spawn(function()
        local count, changed = 0, 0
        for _, root in ipairs(getAllUIRoots()) do
            if not GameFontEnabled then return end
            local ok, descendants = pcall(function() return root:GetDescendants() end)
            if ok and descendants then
                for _, v in ipairs(descendants) do
                    if not GameFontEnabled then return end
                    if trySetFont(v) then changed += 1 end
                    count += 1
                    if count % 300 == 0 then task.wait() end
                end
            end
        end
        print(string.format("[w7w7] Game Font scan: %d objects, %d changed", count, changed))
    end)
end

local function installGameFontLoop()
    if gameFontLoopRunning then return end
    gameFontLoopRunning = true
    task.spawn(function()
        while GameFontEnabled do
            for obj in pairs(gameFontOriginals) do
                if GameFontEnabled and obj.Parent then
                    if gameFontSpec.Face then
                        pcall(function() obj.FontFace = gameFontSpec.Face end)
                    elseif gameFontSpec.Enum then
                        pcall(function() obj.Font = gameFontSpec.Enum end)
                    end
                end
            end
            for _, root in ipairs(getAllUIRoots()) do
                if not GameFontEnabled then break end
                local ok, descendants = pcall(function() return root:GetDescendants() end)
                if ok and descendants then
                    for _, v in ipairs(descendants) do
                        if not GameFontEnabled then break end
                        if not gameFontOriginals[v] then trySetFont(v) end
                    end
                end
            end
            task.wait(1)
        end
        gameFontLoopRunning = false
    end)
end

local function enableGameFont()
    gameFontSpec = getGameFontSpec()
    GameFontEnabled = true
    scanGameFonts()
    if gameFontHookConn then gameFontHookConn:Disconnect() end
    gameFontHookConn = game.DescendantAdded:Connect(function(obj)
        if GameFontEnabled then
            task.defer(function() if GameFontEnabled then trySetFont(obj) end end)
        end
    end)
    installGameFontLoop()
end

local function disableGameFont()
    GameFontEnabled = false
    if gameFontHookConn then gameFontHookConn:Disconnect() gameFontHookConn = nil end
    for obj in pairs(gameFontOriginals) do restoreGameFont(obj) end
    table.clear(gameFontOriginals)
end

local function MakeSlider(label, y, value, minV, maxV, step, onChange)
    local lbl = new("TextLabel", TPOffsetPanel, {
        Size = UDim2.new(1, -20, 0, 14),
        Position = UDim2.new(0, 10, 0, y),
        BackgroundTransparency = 1,
        Text = label,
        TextColor3 = Color3.new(1, 1, 1),
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 41,
    })
    local valLbl = new("TextLabel", TPOffsetPanel, {
        Size = UDim2.new(0, 40, 0, 14),
        Position = UDim2.new(1, -50, 0, y),
        BackgroundTransparency = 1,
        Text = string.format("%.1f", value),
        TextColor3 = Color3.fromRGB(0, 255, 180),
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Right,
        ZIndex = 41,
    })
    local track = new("TextButton", TPOffsetPanel, {
        Size = UDim2.new(1, -20, 0, 8),
        Position = UDim2.new(0, 10, 0, y + 16),
        BackgroundColor3 = Color3.fromRGB(50, 50, 50),
        Text = "",
        ZIndex = 41,
    })
    new("UICorner", track, { CornerRadius = UDim.new(1, 0) })
    local fill = new("Frame", track, {
        Size = UDim2.new(0.5, 0, 1, 0),
        BackgroundColor3 = Color3.fromRGB(0, 140, 60),
        BorderSizePixel = 0,
        ZIndex = 41,
    })
    new("UICorner", fill, { CornerRadius = UDim.new(1, 0) })

    local dragging = false
    local function update(xPos)
        local t = math.clamp((xPos - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local v = minV + (maxV - minV) * t
        v = math.floor(v / step + 0.5) * step
        fill.Size = UDim2.new(t, 0, 1, 0)
        valLbl.Text = string.format("%.1f", v)
        onChange(v)
    end
    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            update(input.Position.X)
        end
    end)
    track.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    track.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            update(input.Position.X)
        end
    end)
    return track
end

local Storage = game:GetService("ReplicatedStorage")
local ItemsModule = Storage:FindFirstChild("Modules")
    and Storage.Modules:FindFirstChild("ItemLibrary")
local itemsTable = nil
if ItemsModule then
    local ok, lib = pcall(function() return require(ItemsModule) end)
    if ok and type(lib) == "table" and type(lib.Items) == "table" then
        itemsTable = lib.Items
    end
end

local GUN_EXCEPTIONS = { Sniper = true, Crossbow = true, Bow = true, RPG = true }
local GUN_ZERO_PROPS = { "ShootSpread", "ShootAccuracy", "ShootRecoil" }
local GUN_FAST_PROPS = { "ShootCooldown", "ShootBurstCooldown" }
local MELEE_PROPS    = { "AttackCooldown", "SwingCooldown", "MeleeCooldown", "Cooldown", "RecoveryTime", "ResetTime" }

local backups = setmetatable({}, { __mode = "k" })

local function isGun(info)
    return type(info) == "table"
        and (info.ShootCooldown ~= nil or info.ShootSpread ~= nil or info.ShootRecoil ~= nil)
end
local function isMelee(info)
    if type(info) ~= "table" or isGun(info) then return false end
    for _, key in ipairs(MELEE_PROPS) do
        if info[key] ~= nil then return true end
    end
    return false
end
local function backup(info, key)
    local saved = backups[info]
    if not saved then saved = {}; backups[info] = saved end
    if saved[key] == nil then saved[key] = info[key] end
end
local function restoreAll()
    for info, saved in pairs(backups) do
        for key, value in pairs(saved) do
            pcall(function() info[key] = value end)
        end
    end
    table.clear(backups)
end
local function applyFastFire()
    if not FastFireEnabled then restoreAll(); return end
    if type(itemsTable) ~= "table" then return end
    for name, info in pairs(itemsTable) do
        if type(info) == "table" then
            if isGun(info) and not GUN_EXCEPTIONS[name] then
                for _, key in ipairs(GUN_ZERO_PROPS) do
                    if info[key] ~= nil then backup(info, key); info[key] = 0 end
                end
                for _, key in ipairs(GUN_FAST_PROPS) do
                    if info[key] ~= nil then backup(info, key); info[key] = FastFireGunRate end
                end
            elseif isMelee(info) then
                for _, key in ipairs(MELEE_PROPS) do
                    if info[key] ~= nil then backup(info, key); info[key] = FastFireMeleeRate end
                end
            end
        end
    end
end

FastFireBtn.MouseButton1Click:Connect(function()
    FastFireEnabled = not FastFireEnabled
    UpdateUI(FastFireBtn, FastFireInd, FastFireEnabled, "Fast Fire")
    if FastFireEnabled then pcall(applyFastFire) else pcall(restoreAll) end
end)

RunService.Heartbeat:Connect(function()
    if KickEnabled then
        pcall(function()
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        hrp.CFrame = TargetPos
                        hrp.AssemblyLinearVelocity = Vector3.zero
                    end
                end
            end
            for p in pairs(Projectiles) do
                if p and p.Parent then
                    p.CFrame = TargetPos
                    p.AssemblyLinearVelocity = Vector3.zero
                else Projectiles[p] = nil end
            end
        end)
    end

    -- TP 黏貼（可調偏移）
    if TpEnabled and currentTarget and currentTarget.Character then
        local targetRoot = currentTarget.Character:FindFirstChild("HumanoidRootPart")
        local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if targetRoot and myRoot then
            -- 構建偏移 CFrame：先轉目標的朝向
            local targetCF = targetRoot.CFrame
            -- 前後（沿目標的 LookVector）、左右（RightVector）、上下（UpVector）
            local offset = CFrame.new(
                TpOffset.Right,    -- X（左右）
                TpOffset.Height,   -- Y（上下）
                -TpOffset.Forward  -- Z（前後，正 = 目標前方）
            )
            myRoot.CFrame = targetCF * offset
            if TpOffset.KeepVelocityZero then
                myRoot.AssemblyLinearVelocity = Vector3.zero
            end
        end
    end

    if YLockEnabled then
        pcall(function()
            local character = LocalPlayer.Character
            if character and character:FindFirstChild("HumanoidRootPart") and character:FindFirstChild("Humanoid") then
                local root = character.HumanoidRootPart
                local humanoid = character.Humanoid
                local currentPos = root.Position
                if not lockedY then lockedY = currentPos.Y - 7 end
                local moveDir = humanoid.MoveDirection
                if moveDir.Magnitude > 0 then
                    local speed = humanoid.WalkSpeed
                    yLastPos = root.Position + (moveDir * speed * 0.016)
                    root.CFrame = CFrame.new(yLastPos.X, lockedY, yLastPos.Z) * (root.CFrame - currentPos)
                else
                    if not yLastPos then yLastPos = currentPos end
                    root.CFrame = CFrame.new(yLastPos.X, lockedY, yLastPos.Z) * (root.CFrame - currentPos)
                end
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
            else
                lockedY = nil
                yLastPos = nil
            end
        end)
    end
end)
